VOICEOVER GENERATOR (ARABIC + URDU) # ========================================== async def generate_all_voiceovers(): print("1/2: Generating clear Arabic & Urdu voiceovers (No Music)...") for num, ar, en, ur in NAMES_99: ar_path = os.path.join(TEMP_AUDIO_DIR, f"{num:02d}_ar.mp3") ur_path = os.path.join(TEMP_AUDIO_DIR, f"{num:02d}_ur.mp3") if not os.path.exists(ar_path): tts_ar = edge_tts.Communicate(ar, ARABIC_VOICE, rate="-10%") await tts_ar.save(ar_path) if not os.path.exists(ur_path): tts_ur = edge_tts.Communicate(ur, URDU_VOICE, rate="-5%") await tts_ur.save(ur_path) # ========================================== # 4. SLIDE IMAGE GENERATOR # ========================================== def prepare_rtl(text): if HAS_RAQM: return text return get_display(reshaper.reshape(text)) def create_slide_image(num, arabic_text, english_text, urdu_text): img = Image.new("RGB", (WIDTH, HEIGHT), color=(10, 38, 28)) draw = ImageDraw.Draw(img) # Gold Decorative Border border_margin = 45 draw.rectangle( [border_margin, border_margin, WIDTH - border_margin, HEIGHT - border_margin], outline=(212, 175, 55), width=6 ) rtl_kwargs = {"direction": "rtl"} if HAS_RAQM else {} # Top Header draw.text((WIDTH // 2, 140), "99 NAMES OF ALLAH | ASMA-UL-HUSNA", font=FONT_HEADER, fill=(212, 175, 55), anchor="mm") # Number Badge (#1 to #99) badge_y = 360 draw.ellipse([WIDTH//2 - 80, badge_y - 80, WIDTH//2 + 80, badge_y + 80], outline=(212, 175, 55), width=5) draw.text((WIDTH // 2, badge_y), f"#{num}", font=FONT_NUMBER, fill=(255, 255, 255), anchor="mm") # Arabic Name draw.text((WIDTH // 2, 720), prepare_rtl(arabic_text), font=FONT_ARABIC, fill=(255, 223, 107), anchor="mm", **rtl_kwargs) # Wrapped English Transliteration & Meaning wrapped_en = "\n".join(textwrap.wrap(english_text, width=28)) draw.multiline_text((WIDTH // 2, 1050), wrapped_en, font=FONT_ENGLISH, fill=(255, 255, 255), anchor="mm", align="center", spacing=14) # Urdu Meaning draw.text((WIDTH // 2, 1340), prepare_rtl(urdu_text), font=FONT_URDU, fill=(180, 240, 200), anchor="mm", **rtl_kwargs) # Footer draw.text((WIDTH // 2, HEIGHT - 130), "SubhanAllah • Alhamdulillah • AllahuAkbar", font=FONT_HEADER, fill=(160, 180, 170), anchor="mm") return np.array(img) # ========================================== # 5. BUILD COMPLETE VIDEO (1 TO 99) # ========================================== def build_complete_video(): asyncio.run(generate_all_voiceovers()) print("2/2: Combining all 99 names and voiceovers into one video...") clips = [] for num, ar, en, ur in NAMES_99: ar_audio = AudioFileClip(os.path.join(TEMP_AUDIO_DIR, f"{num:02d}_ar.mp3")) ur_audio = AudioFileClip(os.path.join(TEMP_AUDIO_DIR, f"{num:02d}_ur.mp3")) # Arabic bole + 0.3s pause + Urdu bole + 0.5s pause from moviepy import concatenate_audioclips combined_audio = concatenate_audioclips([ar_audio, ur_audio]) slide_duration = combined_audio.duration + 0.8 frame = create_slide_image(num, ar, en, ur) clip = ( ImageClip(frame) .with_duration(slide_duration) .with_audio(combined_audio) ) clips.append(clip) final_video = concatenate_videoclips(clips, method="compose") final_video.write_videofile( OUTPUT_VIDEO, fps=FPS, codec="libx264", audio_codec="aac", ffmpeg_params=["-pix_fmt", "yuv420p"] ) print(f"Done! Video saved as: {OUTPUT_VIDEO}") if __name__ == "__main__": build_complete_video()
